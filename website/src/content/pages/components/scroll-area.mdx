```tsx
    <ScrollArea.Thumb />
  </ScrollArea.Scrollbar>
  <ScrollArea.Corner />
</ScrollArea.Root>
```

## Required style

Add these styles to `ScrollArea.Viewport` to hide the native scrollbar:

```css
[data-scroll-area-viewport] {
  scrollbar-width: none;
  &::-webkit-scrollbar {
    display: none;
  }
}
```
